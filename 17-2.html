fetch 获取接口数据

使用原生 fetch() 请求支持跨域访问的公开接口（JSONPlaceholder）， 拿到响应后先用 response.json() 解析成对象，再动态渲染成卡片。全程无任何第三方库。

GET https://jsonplaceholder.typicode.com/users?_limit=10
点击「发起 fetch 请求」加载数据

暂无数据

核心代码（原生 fetch + async/await）

async function czr_load() {
    try {
        var czr_res = await fetch(czr_url);          // 发起请求
        if (!czr_res.ok) {                            // 判断 HTTP 状态码
            throw new Error('HTTP ' + czr_res.status);
        }
        var czr_data = await czr_res.json();          // 把响应体解析成对象
        czr_render(czr_data);                         // 渲染到页面
    } catch (czr_err) {
        czr_showError(czr_err.message);               // 捕获网络 / 解析异常
    }
}
